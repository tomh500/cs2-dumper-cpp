cmake_minimum_required(VERSION 3.20)
project(cs2-dumper VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires 64-bit Windows; configure with -A x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(dumper_core cpp/memory.cpp cpp/pattern.cpp cpp/analysis.cpp cpp/output.cpp)
target_include_directories(dumper_core PUBLIC cpp third_party)
target_compile_definitions(dumper_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(dumper_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
endif()
add_executable(cs2-dumper cpp/main.cpp)
target_link_libraries(cs2-dumper PRIVATE dumper_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(dumper_tests tests/tests.cpp)
  target_link_libraries(dumper_tests PRIVATE dumper_core)
  add_test(NAME core COMMAND dumper_tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "DUMPER_FIXTURES=${CMAKE_CURRENT_SOURCE_DIR}/output")
  add_test(NAME help COMMAND cs2-dumper --help)
endif()
